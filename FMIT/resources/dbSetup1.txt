drop table `flea_market`.`comment`;
drop table `flea_market`.`market`;
drop table `flea_market`.`photo`;
drop table `flea_market`.`user`;
drop table `flea_market`.`marketxref`;
drop table `flea_market`.`item`;

drop schema `flea_market`;

CREATE DATABASE `flea_market`;

CREATE TABLE `flea_market`.`comment` (
  `commentRefId` int NOT NULL AUTO_INCREMENT,
  `comment` varchar(90) DEFAULT NULL,
  PRIMARY KEY (`commentRefId`)
);

CREATE TABLE `flea_market`.`market` (
  `marketRefId` int NOT NULL AUTO_INCREMENT,
  `name` varchar(45) DEFAULT NULL,
  PRIMARY KEY (`marketRefId`)
);

CREATE TABLE `flea_market`.`photo` (
  `photoRefId` int NOT NULL AUTO_INCREMENT,
  `photo` MEDIUMBLOB,
  `itemRefId` int DEFAULT NULL,
  `commentRefId` int DEFAULT NULL,
  PRIMARY KEY (`photoRefId`),
  KEY `itemRefId_idx` (`itemRefId`),
  KEY `photo_comment_idx` (`commentRefId`)
);

CREATE TABLE `flea_market`.`user` (
  `userRefId` int NOT NULL AUTO_INCREMENT,
  `name` varchar(45) DEFAULT NULL,
  `commentRefId` int DEFAULT NULL,
  PRIMARY KEY (`userRefId`),
  KEY `user_comment_idx` (`commentRefId`)
);

CREATE TABLE `flea_market`.`marketxref` (
  `marketXRefId` int NOT NULL AUTO_INCREMENT,
  `marketRefId` int NOT NULL,
  `itemRefId` int NOT NULL,
  PRIMARY KEY (`marketXRefId`),
  KEY `marketRefId_idx` (`marketRefId`),
  KEY `marketItemRefId_idx` (`itemRefId`)
);

CREATE TABLE `flea_market`.`item` (
  `itemRefId` int NOT NULL AUTO_INCREMENT,
  `name` varchar(45) DEFAULT NULL,
  `description` varchar(90) DEFAULT NULL,
  `barCode` varchar(45) DEFAULT NULL,
  `sellerRefId` int NOT NULL,
  `commentRefId` int DEFAULT NULL,
  `boughtDate` date DEFAULT NULL,
  `soldDate` date DEFAULT NULL,
  `boughtPrice` double DEFAULT NULL,
  `soldPrice` double DEFAULT NULL,
  PRIMARY KEY (`itemRefId`),
  UNIQUE KEY `barCode_UNIQUE` (`barCode`),
  KEY `item_seller_idx` (`sellerRefId`),
  KEY `item_comment_idx` (`commentRefId`)
);
